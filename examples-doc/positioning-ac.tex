\usepackage[
  block-width=1.5cm, block-height=1.5cm,

\nskUseModule{*}

\def\c{type=circle, fill=nskMainAccent}
\def\r{type=rounded rectangle, border-radius=0mm, width=3cm, fill=nskBlue}
\def\p{type=polygon[5], fill=nskMainAccent}
\def\t{type=polygon[3], height=2cm, shift-y=-2mm, shift-x=1mm, fill=nskMainAccent}

\def\cp{fill=nskSecondaryAccent!30!nskBg, border-color=nskSecondaryAccent}
\def\x{fill=none, pattern=hatch, border-type=dashed}
\def\o{type=circle, size=5mm, fill=nskRed, shift-x=.5mm}

\def\ph{
  type=rectangle,
  width=1cm,
  height=1cm,
  fill=nskRed,
  border-radius=2mm,
  rotate=0,
  fill=none,
  shadow=false,
  opacity=.5,
  border-type=dashed,
}

\def\guide{color=nskStrongCyan!85!black, arrow-type=dashed}

\begin{document}
\begin{nskFigure}[]

	{

	\nskBlock*[last-pos={right=7mm}, \t]
	}

		\nskBlock*[\o, last-pos={above=7mm}]

	\nskConnect[

	}



	\nskEdgeFade[edge=up]
