\begin{document}

  \foreach \s/\l in {rectangle/1cm, circle/1.5cm, diamond/1.5cm} {

      \foreach \p[count=\i] in {0,.1,...,1} {

      \nskMarkPath[at=0]{ref}{\nskMark[id=p0]}

      ]

    }
  }
